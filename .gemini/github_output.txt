14 | async function postToGAS(url, payload) {
15 |     try {
16 |         payload.authKey = "JUWA-Auth-Key";
17 |         const response = await fetch(url, {
18 |             method: 'POST',
19 |             // GASのdoPostで確実にパースさせるための設定
20 |             headers: {
21 |                 'Content-Type': 'text/plain'
22 |             },
23 |             // ここで一括してJSON文字列に変換
24 |             body: JSON.stringify(payload)
25 |         });
26 | 
27 |         if (!response.ok) {
28 |             throw new Error(`HTTP error! status: ${response.status}`);
29 |         }
30 | 
31 |         const result = await response.json();
32 |         
33 |         if (result.status === "error") {
34 |             throw new Error(result.message || "GAS実行エラー");
35 |         }
36 | 
37 |         return result;
38 |     } catch (e) {
39 |         console.error("GAS送信失敗:", e);
40 |         throw e; // 呼び出し元で個別のエラー表示を行うために再スロー
41 |     }

228 | /**
229 |  * スコアボードUI更新
230 |  * @returns {void}
231 |  */
232 | function updateScoreboardUI() {
233 |     // 1. 要素を取得
234 |     const headerRow = document.getElementById('header-row');
235 |     const topRow = document.getElementById('score-row-top');
236 |     const bottomRow = document.getElementById('score-row-bottom');
237 | 
238 |     // 2. ここでチェック！どれかが null ならエラーメッセージを出して中断する
239 |     if (!headerRow || !topRow || !bottomRow) {
240 |         console.error("エラー: スコアボードの行が見つかりません。", {headerRow, topRow, bottomRow});
241 |         return; 
242 |     }
243 | 
244 |     // --- 以降、安全に実行される ---
245 |     const clearInningCells = (row) => {
246 |         // row が undefined でないことは上記で確認済みなので length は安全に読める
247 |         for (let i = row.cells.length - 2; i >= 1; i--) {
248 |             row.deleteCell(i);
249 |         }
250 |     };
251 | 
252 |     clearInningCells(headerRow);
253 |     clearInningCells(topRow);
254 |     clearInningCells(bottomRow);
255 | 
256 |     const displayInnings = Math.max(currentInning, totalInnings);
257 | 
258 |     for (let i = 1; i <= displayInnings; i++) {
259 |         // ヘッダー挿入
260 |         const th = document.createElement('th');
261 |         th.textContent = i;
262 |         headerRow.insertBefore(th, headerRow.cells[headerRow.cells.length - 1]);
263 | 
264 |         // 先攻セル挿入
265 |         const tdTop = document.createElement('td');
266 |         tdTop.textContent = (score.top && score.top[i] !== undefined) ? score.top[i] : "-";
267 |         topRow.insertBefore(tdTop, topRow.cells[topRow.cells.length - 1]);
268 | 
269 |         // 後攻セル挿入
270 |         const tdBottom = document.createElement('td');
271 |         tdBottom.textContent = (score.bottom && score.bottom[i] !== undefined) ? score.bottom[i] : "-";
272 |         bottomRow.insertBefore(tdBottom, bottomRow.cells[bottomRow.cells.length - 1]);
273 | 
274 |         // アクティブ表示（点滅）
275 |         if (i === currentInning) {
276 |             if (!isBottomInning) tdTop.classList.add('active-cell');
277 |             else tdBottom.classList.add('active-cell');
278 |         }
279 |     }
280 | 
281 |     // 合計点の書き換え
282 |     const topTotalEl = document.getElementById('total-score-top');
283 |     const bottomTotalEl = document.getElementById('total-score-bottom');
284 |     if (topTotalEl) topTotalEl.textContent = totalScore.top;
285 |     if (bottomTotalEl) bottomTotalEl.textContent = totalScore.bottom;
286 | }

 345 | /**
 346 |  * 打撃結果の処理
 347 |  * @param {string} actionName - "シングルヒット", "四球" など
 348 |  */
 349 | async function recordPlay(actionName) {
 350 |     if (isGameEnded || isPushing) return;
 351 |      
 352 |     setControlsDisabled(true);
 353 |     try{
 354 |     if (actionName === "ホームラン") {
 355 |         let runs = 1;
 356 |         if (runners.base1) runs++;
 357 |         if (runners.base2) runs++;
 358 |         if (runners.base3) runs++;
 359 |         addScore(runs);
 360 |         runners = { base1: false, base2: false, base3: false };
 361 |     } 
 362 |     else if (actionName === "シングルヒット") {
 363 |         if (runners.base3) addScore(1);
 364 |         runners.base3 = runners.base2;
 365 |         runners.base2 = runners.base1;
 366 |         runners.base1 = true;
 367 |     } 
 368 |     else if (actionName === "ダブルヒット") {
 369 |         if (runners.base3) addScore(1);
 370 |         if (runners.base2) addScore(1);
 371 |         runners.base3 = runners.base1;
 372 |         runners.base2 = true;
 373 |         runners.base1 = false;
 374 |     } 
 375 |     else if (actionName === "トリプルヒット") {
 376 |         let runs = 0;
 377 |         if (runners.base1) runs++;
 378 |         if (runners.base2) runs++;
 379 |         if (runners.base3) runs++;
 380 |         addScore(runs);
 381 |         runners = { base1: false, base2: false, base3: true };
 382 |     }
 383 |     // --- 追加：四球（押し出し）のロジック ---
 384 |     else if (actionName === "四球") {
 385 |         if (runners.base1 && runners.base2 && runners.base3) {
 386 |             // 満塁なら押し出し得点
 387 |             addScore(1);
 388 |         } else if (runners.base1 && runners.base2) {
 389 |             // 1,2塁なら3塁へ進塁
 390 |             runners.base3 = true;
 391 |         } else if (runners.base1) {
 392 |             // 1塁のみなら2塁へ進塁
 393 |             runners.base2 = true;
 394 |         }
 395 |         // どんな状況でも1塁は埋まる
 396 |         runners.base1 = true;
 397 |     }
 398 | 
 399 |     // カウントリセット
 400 |     counts.strike = 0;
 401 |     counts.ball = 0;
 402 |     
 403 |     await syncPush(actionName, getLogSnapshot());
 404 |     if(isBottomInning) {
 405 |         battingCount.bottom ++;
 406 |     }else {
 407 |         battingCount.top ++;
 408 |     }     
 409 |     updateCountDisplay();
 410 |     updateDiamondDisplay();
 411 |     updateScoreboardUI();
 412 |     updateShareLinkDisplay()
 413 |     historyStack = [];
 414 | 
 415 |     }   
 416 |     catch(error) {
 417 |         console.error("送信エラー:", error);
 418 |         alert("データの送信に失敗しました。通信環境を確認してください。");
 419 |         undo();
 420 |     }finally {
 421 | 
 422 |         setControlsDisabled(false);
 423 |     }
 424 | }
